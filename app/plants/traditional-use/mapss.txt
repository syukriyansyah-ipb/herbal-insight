'use client'
import React, { Fragment, useEffect, useRef, useState } from 'react'
import maplibregl from 'maplibre-gl';
import { getTradisionalUse } from '@/api/reasonings';


const Maps = ({ plantName }) => {

    const mapElementRef = useRef(null);
    const contentRef = useRef(null);
    const showMap = async () => {
        const map = new maplibregl.Map({
            container: mapElementRef.current,
            style: 'https://api.maptiler.com/maps/streets/style.json?key=RFLeqoTvQAwwRUw6Ozzv',
            bounds: [
                [106.14177, -7.703153],
                [108.905747, -5.87488],
            ],
            zoom: 4,
            maxZoom: 20,
            minZoom: 3,
        });
        // if(data){
        const response = await getTradisionalUse(plantName);
        console.log('data respon', response?.data);
        const dataMap = response?.data?.map((item) => {
            return {
                type: 'Feature',
                properties: {
                    // 'description' : '<strong>dsadasdasdasdasdasdassadsad</strong>'
                    description: `<p>zzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzPlant Part : ${item.plant_part?.properties?.label} <br /> Traditional Use : ${item.traditional_use?.properties?.label} <br /> Method of Use : ${item.method_of_use?.properties?.label}</p>`,
                    // // id: index,
                    // data : {
                    //     plant_part: item.plant_part?.properties?.label,
                    //     traditional_use: item.traditional_use?.properties?.label,
                    //     method_of_use: item.method_of_use?.properties?.label
                    // }
                },
                geometry: {
                    type: 'Point',
                    coordinates: [parseFloat(item.location.properties.long), parseFloat(item.location.properties.lat)]
                },
            }
        })
        console.log('dataMap', dataMap);

        map.on('load', async () => {
            const image = await map.loadImage('https://maplibre.org/maplibre-gl-js/docs/assets/custom_marker.png');
            // Add an image to use as a custom marker
            map.addImage('custom-marker', image.data);
            // Add a GeoJSON source with 15 points
            map.addSource('places', {
                'type': 'geojson',
                'data': {
                    'type': 'FeatureCollection',
                    'features': dataMap
                }
            });

            map.addLayer({
                'id': 'places',
                'type': 'symbol',
                'source': 'places',
                'layout': {
                    'icon-image': 'custom-marker',
                    "icon-overlap": "always"
                }
            });
            // Create a popup, but don't add it to the map yet.
            const popup = new maplibregl.Popup({
                closeButton: false,
                closeOnClick: false
            });
            map.on('mouseenter', 'places', (e) => {
                // Change the cursor style as a UI indicator.
                map.getCanvas().style.cursor = 'pointer';

                const coordinates = e.features[0].geometry.coordinates.slice();
                const description = e.features[0].properties.description;
                console.log('desc', description);

                // Ensure that if the map is zoomed out such that multiple
                // copies of the feature are visible, the popup appears
                // over the copy being pointed to.
                while (Math.abs(e.lngLat.lng - coordinates[0]) > 180) {
                    coordinates[0] += e.lngLat.lng > coordinates[0] ? 360 : -360;
                }

                // Populate the popup and set its coordinates
                // based on the feature found.
                popup.setLngLat(coordinates).setHTML(description).addTo(map);
                // popup.setLngLat(coordinates).setDOMContent(contentRef.current).setMaxWidth('400px').addTo(map);
            });

            map.on('mouseleave', 'places', () => {
                map.getCanvas().style.cursor = '';
                popup.remove();
            });
        });
        // }


    }
    useEffect(() => {
        showMap()

    }, [plantName])
    return (
        <Fragment>
            <div ref={mapElementRef} className='w-full h-[400px] relative overflow-hidden'>

            </div>
            <div className='hidden'>
            <div ref={contentRef}>
            <p>sjfsafasfkmadkfna</p>
            <p>sjfsafasfkmadkfna</p>
            <p>sjfsafasfkmadkfna</p>
            <p>sjfsafasfkmadkfna</p>
            <p>sjfsafasfkmadkfna</p>
            <p>sjfsafasfkmadkfna</p>
                
            </div>
            </div>
           

        </Fragment>
    )
}

export default Maps
