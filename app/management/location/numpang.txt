'use client'
import React, { Fragment, useRef } from "react";
import logoTU from "@/public/assets/icons/iconTradisionalUses.svg"
import Image from "next/image";
import { IoSearch } from "react-icons/io5";
import { IoAddSharp } from 'react-icons/io5';
import { FaRegEdit } from "react-icons/fa";
import { LuEye } from "react-icons/lu";
import { MdOutlineDeleteForever } from "react-icons/md";
import { GrHomeRounded } from "react-icons/gr";
import { useState } from "react";
import { LiaUserEditSolid } from "react-icons/lia";
import useSWR from "swr";
import { fetcher } from "@/helpers/common";
import ViewLocation from "./ViewLocation";
import FormLocation from "./FormLocation";
import UpdateLocation from "./UpdateLocation";

const Location = () => {
  const [currentPage, setCurrentPage] = useState(1);
  const pageSize = 10;

  const refLabel = useRef();
  const [search, setSearch] = useState({
    label: '',
  })

  const { data, error, mutate, isLoading } = useSWR(`/location/?skip=${(currentPage - 1) * pageSize}&limit=${pageSize}`, fetcher);
  const handleChangeSearch = (e) => {
    console.log(e)
    if (e.key === 'Enter' || e.target.value == '') {
      setSearch((current) => ({
        ...current,
        label: refLabel.current.value,
      }))
    }
  }

  const handleSearch = () => {
    setSearch((current) => ({
      ...current,
      label: refLabel.current.value,
    }))
    setCurrentPage(1)
  }


  const [showView, setShowView] = useState(false);
  const [viewItem, setViewItem] = useState(null);
  const handleViewDetails = (data) => {
    setViewItem(data);
    setShowView(true);
  };
  const closeViewDetails = () => {
    setShowView(false);
  }


  const [modalUpdate, setModalUpdate] = useState({
    show: false,
    data: null
  });
  const handleSelectUpdate = (data) => {
    setModalUpdate({
      show: true,
      data: data
    })
  }


  const handleNextPage = () => {
    setCurrentPage(currentPage + 1);
  }

  const handlePreviousPage = () => {
    setCurrentPage(currentPage - 1);
  }

  const [showPopup, setShowPopup] = useState(false);

  const handlePopupConfirm = () => {
    setShowPopup(false);
    window.location.href = "/";
  }

  const [showEdit, setShowEdit] = useState(false);
  const [iconSize, setIconSize] = useState(30);

  const toggleEdit = () => {
    setShowEdit(!showEdit); // Mengubah nilai showEdit menjadi kebalikannya
    if (!showEdit) {
      setIconSize(40);
    } else {
      setIconSize(30);
    }
  };

  const [showForm, setShowForm] = useState(false);
  return (
    <Fragment className=" bg-gray-100">
      <div className="flex mb-4 pt-6 ml-5 justify-between">
        <div className="flex ">
          <Image src={logoTU} alt="logoPN" className="w-10 h-10 mr-4" />
          <h1 className="text-3xl font-bold"> Manage Location</h1>
        </div>
        <GrHomeRounded className="ml-auto mr-10 mt-2 cursor-pointer" size={25} onClick={() => setShowPopup(true)} title="Homepage" />

      </div>
      {/* <div className="relati"> */}
      <div className="fixed bottom-0 left-0 bg-gray-300 rounded-full pl-3 pb-2 ml-5 mb-5">
        <LiaUserEditSolid className="ml-auto mr-2 mt-2 cursor-pointer " size={iconSize} onClick={toggleEdit} title="Edit Profile" />
        {showEdit && (
          <div className="absolute bottom-[7px] left-[65px] h-10 cursor-pointer hover:bg-gray-300  bg-gray-200 p-3 rounded-r-full shadow-lg w-32">
            <h1 className="text-sm font-semibold">Edit Profiles</h1>
          </div>
        )}
      </div>
      {/* </div> */}
      {showPopup && (
        <div className="fixed inset-0 flex items-center justify-center bg-gray-600 bg-opacity-50 z-50">
          <div className="bg-white p-8 rounded-lg shadow-lg relative">
            <p className="text-lg font-semibold mb-4">Are you sure you want to log out?</p>
            <div className="flex justify-end">
              <button onClick={() => setShowPopup(false)} className="bg-gray-300 px-4 py-2 rounded mr-2">NO</button>
              <button onClick={handlePopupConfirm} className="bg-blue-500 text-white px-4 py-2 rounded">YES</button>
            </div>
          </div>
        </div>
      )}
      <div className="bg-white rounded-xl mx-5 pb-96">
        <div className="flex items-center justify-between gap-3 mb-5 ml-5 mr-5 pt-3">
          <div className='relative flex flex-1'>
            <input type='text' ref={refLabel} onKeyDown={handleChangeSearch} className='w-full border border-gray-200 py-3 pl-5 text-sm outline-2 rounded-xl bg-[#EFEFEF]' placeholder='Search for traditional uses by name...' />
            <IoSearch className='absolute right-3 top-3 h-6 w-6 text-[#636363]' onClick={handleSearch} />
          </div>

          <button href="#" className='inline-flex items-center space-x-1 text-[#2e2e2e] bg-[#EFEFEF] hover:bg-[#D1D1D1] hover:ring-2 hover:ring-black px-5 py-3 rounded-xl text-sm border border-gray-200'
            onClick={ () => setShowForm(true)}>
            <IoAddSharp size={20} className='mr-1' />
            New
          </button>
        </div>

        <div className="px-5">
          <table className="w-full text-sm text-left text-gray-500 border-separate border-spacing-y-2 table-a overflow-scroll">
            <thead className="h-[50px] bg-[#50744C] text-slate-100 text-[14px]">
              <tr className="">
                <th className="rounded-l-xl pl-3">Location (ID)</th>
                <th className="pl-3">Kelurahan</th>
                <th className="pl-3">Kecamatan</th>
                <th className="pl-3">Kabupaten</th>
                <th className="pl-3">Provinsi</th>
                <th className="pl-3">Lintang</th>
                <th className="pl-3">Bujur</th>
                <th className="pl-3">User ID</th>
                <th className="rounded-r-xl text-center">Action</th>
              </tr>
            </thead>
            <tbody className="text-black ">
              {data?.data?.map((item) => (
                <tr className="text-[13px] text-left " key={item.id}>
                  <td className="border-t border-b border-l border-[#BDBDBD] pl-3 pt-2 pb-2 rounded-l-xl ">
                    <p>{item.Location.properties.id || "-"}</p>
                  </td>
                  <td className="border-t border-b  border-[#BDBDBD] pl-3 pt-2 pb-2 ">
                    <p>{item.Location.properties.kecamatan || "-"}</p>
                  </td>
                  <td className="border-t border-b  border-[#BDBDBD] pl-3 pt-2 pb-2 ">
                    <p>{item.Location.properties.kel_desa || "-"}</p>
                  </td>
                  <td className="border-t border-b  border-[#BDBDBD] pl-3 pt-2 pb-2 ">
                    <p>{item.Location.properties.kab_kota || "-"}</p>
                  </td>
                  <td className="border-t border-b  border-[#BDBDBD] pl-3 pt-2 pb-2 ">
                    <p>{item.Location.properties.provinsi || "-"}</p>
                  </td>
                  <td className="border-t border-b  border-[#BDBDBD] pl-3 pt-2 pb-2 ">
                    <p>{item.Location.properties.lat || "-"}</p>
                  </td>
                  <td className="border-t border-b  border-[#BDBDBD] pl-3 pt-2 pb-2 ">
                    <p>{item.Location.properties.long || "-"}</p>
                  </td>
                  <td className="border-t border-b  border-[#BDBDBD] pl-3 pt-2 pb-2 ">
                    <p>{item.User.properties.id || "-"}</p>
                  </td>
                  <td className="border-t border-b border-r border-[#BDBDBD] pl-3 pt-2 pb-2 rounded-r-xl">
                    <div className="flex gap-2 mr-2 justify-center items-center">
                      <LuEye size={23} className="cursor-pointer ml-0.5" onClick={() => handleViewDetails(item)} />
                      <FaRegEdit size={20} onClick={() => handleSelectUpdate(item)} className="cursor-pointer ml-1" />
                      <MdOutlineDeleteForever size={24} className="mt-1"/>
                    </div>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
          <div className="mt-4 flex justify-between ">
            {currentPage > 1 && (
              <button
                className="bg-[#EFEFEF] hover:bg-[#D1D1D1] hover:ring-2 hover:ring-black px-5 py-3 rounded-xl text-sm border font-semibold border-gray-200"
                onClick={handlePreviousPage}
              >
                Previous
              </button>
            )}
            {data && (
              <button
                className="bg-[#EFEFEF] hover:bg-[#D1D1D1] hover:ring-2 hover:ring-black px-5 py-3 rounded-xl text-sm border font-semibold border-gray-200"
                onClick={handleNextPage}
              >
                Next
              </button>
            )}
          </div>
        </div>
      </div>
      {showView && (
        <ViewLocation isVisible={showView} onClose={closeViewDetails} data={viewItem} />
      )}
      <FormLocation isVisible={showForm} onClose={() => setShowForm(false)} />
      {modalUpdate && modalUpdate.show && (
        <UpdateLocation
          isVisible={modalUpdate.show}
          data={modalUpdate.data}
          onClose={() => setModalUpdate({ show: false, data: null })}
        />
      )}
    </Fragment>
  )
}

export default Location
